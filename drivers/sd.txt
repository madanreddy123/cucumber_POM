public void verifyPaymentRequestResponse(String serviceName, String periodId) {

    try {
        int adjustmentPortfolioCount_DB = 0;

        Type portfolioAdjustment_Type =
                new TypeToken<ArrayList<PortfolioAdjustment>>() {}.getType();

        sql.getConnection();

        ResultSet portfolioAdjusted = null;

        switch (serviceName) {

            case "PortfolioAdjustment":
                portfolioAdjusted = sql.executeQuery(
                        "select count(*) from COMN_USES.CU_PORTFOLIO_ADJUSTMENT " +
                        "where cu_period_id = '" + periodId + "'"
                );
                break;

            default:
                Assert.fail("Unsupported serviceName: " + serviceName);
                break;
        }

        while (portfolioAdjusted.next()) {
            adjustmentPortfolioCount_DB =
                    portfolioAdjusted.getInt("COUNT(*)");
        }

        ArrayList<PortfolioAdjustment> portfolioAdjustments =
                gson.fromJson(
                        apiResponse.getBody().asString(),
                        portfolioAdjustment_Type
                );

        ArrayList<Integer> cuPortfolioIDIDList = new ArrayList<>();

        for (int i = 0; i < portfolioAdjustments.size(); i++) {
            cuPortfolioIDIDList.add(
                    portfolioAdjustments.get(i).getCUPortfolioId()
            );
        }

        Assert.assertEquals(
                adjustmentPortfolioCount_DB,
                cuPortfolioIDIDList.size()
        );

    } catch (SQLException throwable) {
        throwable.printStackTrace();
        Assert.fail("SQL execution failed: " + throwable.getMessage());

    } finally {
        sql.closeConnection();
    }
}
